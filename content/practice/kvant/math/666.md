import sys
input = sys.stdin.readline

def solve():
    t = int(input())
    for _ in range(t):
        n, m, p = map(int, input().split())
        s = list(map(lambda x: int(x)-1, input().split()))
        INF = 10**18
        max_edge = [[INF]*n for _ in range(n)]
        for i in range(n):
            max_edge[i][i] = 0
        for _ in range(m):
            u, v, w = map(int, input().split())
            u -= 1
            v -= 1
            max_edge[u][v] = w
            max_edge[v][u] = w

        # Floyd-Warshall for minimal max-edge paths
        for k in range(n):
            for i in range(n):
                for j in range(n):
                    max_edge[i][j] = min(max_edge[i][j], max(max_edge[i][k], max_edge[k][j]))

        # Compute minimal total latency for each number of servers
        min_latency_per_house = [INF]*p
        result = []
        used_servers = set()
        for _ in range(n):
            # Choose the next best server to minimize sum
            best_improvement = -1
            best_house = -1
            for v in range(n):
                if v in used_servers:
                    continue
                improvement = 0
                for idx, house in enumerate(s):
                    improvement += max(0, min_latency_per_house[idx] - max_edge[house][v])
                if improvement > best_improvement:
                    best_improvement = improvement
                    best_house = v
            used_servers.add(best_house)
            for idx, house in enumerate(s):
                min_latency_per_house[idx] = min(min_latency_per_house[idx], max_edge[house][best_house])
            result.append(str(sum(min_latency_per_house)))
        print(' '.join(result))

if __name__ == "__main__":
    solve()

## 问题理解

 该问题考虑了一个网络$n$连接的顶点$m$具有正整数权重的无向边。 的一个子集$p$顶点代表需要最小延迟连接到一组服务器的房屋。 房屋之间的延迟$s_i$和服务器$v$定义为沿任何连接路径的最小最大边权重$s_i$和$v$，这称为 **瓶颈距离**$d(s_i, v)$。 对于每个整数$k$从$1$到$n$，任务是计算所有房屋的最小可能的延迟总和，如果恰好$k$服务器已安装。 正式来说，为了$k \in [1, n]$, 必须确定$$\operatorname{OPT}(k) = \min_{S \subseteq V, |S|=k} \sum_{i=1}^{p} \min_{v \in S} d(s_i, v),$$在哪里$S$是服务器位置的集合。 该解决方案必须保证每个问题的全局最优性$k$; 任何不提供这种保证的启发都是不够的。 关键的挑战在于枚举或以其他方式确定子集$S$从而达到$\operatorname{OPT}(k)$不依赖于未经证实的假设，例如贪婪最优性。 

## 瓶颈距离计算

 每对顶点之间的最小最大边路径可以使用改进的 Floyd-Warshall 算法来计算。 定义一个矩阵$d[i][j]$初始化为$0$直接连接顶点的对角线和边权重，而非相邻顶点设置为$+\infty$。 更新规则$$d[i][j] = \min(d[i][j], \max(d[i][k], d[k][j]))$$正确捕获穿过中间顶点的路径的最小可能最大边权重$k$。 迭代所有$k, i, j$为了确保完成后，$d[i][j]$等于顶点之间的瓶颈距离$i$和$j$。 该计算是精确的，不需要近似，时间复杂度为$O(n^3)$。 

## 精确优化方法

 每个服务器的选择问题$k$是组合的并且可以精确地求解$n$通过枚举所有子集$k$顶点。 让$S$是大小的候选子集$k$。 对于每个房子$s_i$, 计算$L_i = \min_{v \in S} d(s_i, v)$，然后将所有房屋的这些值相加以获得总延迟。 跟踪所有大小子集的最小总和$k$确定$\operatorname{OPT}(k)$。 这保证了全局最优性，因为所有子集都被考虑。 对于较大的$n$，额外的算法结构（例如子集上的动态规划或整数线性规划）可能是必要的，但对于任何精确的解决方案，关键原则是必须使用严格证明的数学属性来评估或证明减少服务器位置的每种可能的组合。 

## Python 实现

 以下 Python 3 解决方案使用修改后的 Floyd–Warshall 方法计算精确的最小总延迟，并枚举大小的所有子集$k$以获得全局最优性。 它采用适合多个测试用例的快速输入读取。```
import sys
import itertools
input = sys.stdin.readline

def solve():
    t = int(input())
    for _ in range(t):
        n, m, p = map(int, input().split())
        houses = list(map(lambda x: int(x)-1, input().split()))
        INF = 10**18
        d = [[INF]*n for _ in range(n)]
        for i in range(n):
            d[i][i] = 0
        for _ in range(m):
            u, v, w = map(int, input().split())
            u -= 1
            v -= 1
            d[u][v] = w
            d[v][u] = w

        for k in range(n):
            for i in range(n):
                for j in range(n):
                    d[i][j] = min(d[i][j], max(d[i][k], d[k][j]))

        results = []
        vertices = list(range(n))
        for k in range(1, n+1):
            min_total = INF
            for subset in itertools.combinations(vertices, k):
                total = 0
                for house in houses:
                    total += min(d[house][v] for v in subset)
                min_total = min(min_total, total)
            results.append(str(min_total))
        print(' '.join(results))
