[欧拉项目 972 问题](https://projecteuler.net/problem=972)

 ## 解决方案

 **答案：**`3575508`让$$s(P)=x^2+y^2+1.$$庞加莱圆盘中的双曲测地线是：

 - 通过原点的直径，或
 - 与单位圆正交的圆。 

这样的圆有方程$$x^2+y^2+Ax+By+1=0.$$因此三点$P_i=(x_i,y_i)$位于共同的测地线上 iff 存在$A,B$这样$$x_i^2+y_i^2+A x_i+B y_i+1=0$$为所有人$i$。 消除$A,B$给出行列式条件$$\det \begin{pmatrix} x_1 & y_1 & x_1^2+y_1^2+1\\ x_2 & y_2 & x_2^2+y_2^2+1\\ x_3 & y_3 & x_3^2+y_3^2+1 \end{pmatrix}=0.$$所以每个点对应一个投影点$$(x,y,x^2+y^2+1),$$和测地线与投影线完全对应。 

计算策略为：

 1. 生成带分母的单位圆盘中的所有有理点$\le 12$。 
2. 将每个点转换为整数射影向量。 
3. 每对都确定一条唯一的投影线（通过叉积）。 
4. 计算每条线上有多少个点。 
5. 总和$k(k-1)(k-2)$在所有行上，因为需要有序的三元组。 

以下 C++ 程序执行精确计算：```python
#include <bits/stdc++.h>
using namespace std;
using ll = long long;

struct Key {
    ll a,b,c;
};

struct H {
    size_t operator()(Key const& k) const noexcept {
        return ((uint64_t)k.a * 1315423911u)
             ^ ((uint64_t)k.b << 21)
             ^ k.c;
    }
};

struct E {
    bool operator()(Key const& x, Key const& y) const noexcept {
        return x.a==y.a && x.b==y.b && x.c==y.c;
    }
};

ll gcd3(ll a,ll b,ll c){
    return gcd(gcd(llabs(a), llabs(b)), llabs(c));
}

int main() {
    int N = 12;

    vector<pair<int,int>> vals;
    set<pair<int,int>> seen;

    // all reduced rationals with denominator <= N
    for(int q=1;q<=N;q++){
        for(int p=-q+1;p<q;p++){
            int g = gcd(abs(p), q);
            pair<int,int> t = {p/g, q/g};
            if(!seen.count(t)){
                seen.insert(t);
                vals.push_back(t);
            }
        }
    }

    // projective points
    vector<array<int,3>> pts;

    for(auto [p,q] : vals){
        for(auto [r,s] : vals){

            if(1LL*p*p*s*s + 1LL*r*r*q*q < 1LL*q*q*s*s){

                int g = gcd(q,s);
                int l = q/g*s;

                ll X = 1LL*p*(l/q);
                ll Y = 1LL*r*(l/s);

                ll a = X*l;
                ll b = Y*l;
                ll c = X*X + Y*Y + 1LL*l*l;

                ll gg = gcd3(a,b,c);

                pts.push_back({
                    (int)(a/gg),
                    (int)(b/gg),
                    (int)(c/gg)
                });
            }
        }
    }

    unordered_map<Key,int,H,E> mp;
    mp.reserve(30000000);

    int n = pts.size();

    // each pair determines a line
    for(int i=0;i<n;i++){
        auto [a1,b1,c1] = pts[i];

        for(int j=i+1;j<n;j++){

            auto [a2,b2,c2] = pts[j];

            ll A = 1LL*b1*c2 - 1LL*c1*b2;
            ll B = 1LL*c1*a2 - 1LL*a1*c2;
            ll C = 1LL*a1*b2 - 1LL*b1*a2;

            ll g = gcd3(A,B,C);

            A/=g; B/=g; C/=g;

            if(A<0 || (A==0 && (B<0 || (B==0 && C<0)))){
                A=-A; B=-B; C=-C;
            }

            mp[{A,B,C}]++;
        }
    }

    long long ans = 0;

    // if a line has k points, it contributes k(k-1)(k-2)
    for(auto &kv : mp){

        long long v = kv.second;

        // v = C(k,2)
        long long s = sqrtl(1 + 8*v);
        long long k = (1 + s)/2;

        ans += k*(k-1)*(k-2);
    }

    cout << ans << endl;
}
